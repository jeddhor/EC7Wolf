defaultmap
{
	music = "C7MUS00"
	victorynext = "EndTitle"
	bordertexture = "#000000"
	// The white/purple sector, which is every level except the eight overridden
	// below. Corridor 7 keeps these per level in its executable, not in the map
	// data -- every shipped map's flats plane is uniformly zero, there are no
	// floor/ceiling tags, and the level header is fully accounted for by
	// offsets/lengths/dimensions/name with no spare bytes. Values are palette
	// indices 39 (white, the top of the grey ramp 16-39) and 199 (mid the
	// purple ramp 184-207). Index 15 is also pure white but is a standalone
	// entry, not the head of a ramp -- shading down from it walks into the
	// skin tones at 14 and below. Written as "@NNN"
	// (exact palette entry) rather than "#RRGGBB": the plane shading walks the
	// palette by index, and Corridor 7 repeats colours across ramps, so a
	// colour-matched flat can land in the wrong ramp and fade out the wrong way.
	defaultfloor = "@199"
	defaultceiling = "@39"
	// Corridor 7 uses a full-range depth ramp:
	// MAP01's nearby surfaces stay bright while its distant fence line and
	// floor/ceiling fade toward the dark palette ramps.
	DefaultLighting = 130
	DefaultVisibility = 10.0
	DefaultMaxLightVis = 30.0
}
